cmake_minimum_required(VERSION 3.25)
project(GUIFormsRejectMixedCore LANGUAGES CXX)
find_package(GUIForms CONFIG REQUIRED COMPONENTS Core Application)
add_executable(mixed_core main.cpp)
target_link_libraries(mixed_core PRIVATE GUIForms::Application)
if(LINK_CASE STREQUAL "direct")
    target_link_libraries(mixed_core PRIVATE GUIForms::Core)
elseif(LINK_CASE STREQUAL "static_adapter")
    add_library(adapter STATIC main.cpp)
    target_link_libraries(adapter PRIVATE GUIForms::Core)
    target_link_libraries(mixed_core PRIVATE adapter)
elseif(LINK_CASE STREQUAL "interface_adapter")
    add_library(adapter INTERFACE)
    target_link_libraries(adapter INTERFACE GUIForms::Controls)
    target_link_libraries(mixed_core PRIVATE adapter)
else()
    message(FATAL_ERROR "Unknown rejection fixture")
endif()
