cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(stx_conv VERSION 0.1.0 LANGUAGES CXX)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(STX_CONV_TESTS_DEFAULT ON)
else()
    set(STX_CONV_TESTS_DEFAULT OFF)
endif()
option(STX_CONV_TESTS "Build deterministic algorithm tests" ${STX_CONV_TESTS_DEFAULT})
option(STX_CONV_SANITIZE "Instrument library and tests" OFF)
set(STX_CONV_ORACLE "" CACHE PATH "Pinned Plan Paint test-only source directory")
add_library(stx_conv STATIC src/plan.cpp src/profile.cpp src/image.cpp)
add_library(stx_conv::stx_conv ALIAS stx_conv)
target_compile_features(stx_conv PUBLIC cxx_std_20)
set_target_properties(stx_conv PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_include_directories(stx_conv PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(stx_conv PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -ffp-contract=off)
endif()
if(STX_CONV_SANITIZE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "This sanitizer configuration needs Clang or GCC")
    endif()
    # Every undefined-behaviour report stops the test, as in the repository's
    # sanitizer build (tools/ci_build.py).
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    # The directory options reach only targets created after this point.
    target_compile_options(stx_conv PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
endif()
if(STX_CONV_TESTS)
    enable_testing()
    add_executable(stx_conv_tests tests/core.cpp)
    target_link_libraries(stx_conv_tests PRIVATE stx_conv::stx_conv)
    add_test(NAME stx_conv_core COMMAND stx_conv_tests)
    add_executable(stx_conv_precision tests/precision.cpp)
    target_link_libraries(stx_conv_precision PRIVATE stx_conv::stx_conv)
    add_test(NAME stx_conv_precision COMMAND stx_conv_precision)
    add_executable(stx_conv_sweep tests/sweep.cpp)
    target_link_libraries(stx_conv_sweep PRIVATE stx_conv::stx_conv)
    add_test(NAME stx_conv_sweep COMMAND stx_conv_sweep)
    add_executable(stx_conv_fuzz_harness fuzz/conv_fuzz.cpp)
    target_link_libraries(stx_conv_fuzz_harness PRIVATE stx_conv::stx_conv)
    target_compile_definitions(stx_conv_fuzz_harness PRIVATE STX_CONV_FUZZ_STANDALONE=1)
    add_test(NAME stx_conv_fuzz_harness COMMAND stx_conv_fuzz_harness)
    if(STX_CONV_ORACLE)
        include(cmake/verify_oracle.cmake)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_float_oracle.py"
            "${STX_CONV_ORACLE}/src/conv.cpp" "${CMAKE_CURRENT_BINARY_DIR}/float_oracle.hpp"
            COMMAND_ERROR_IS_FATAL ANY)
        add_executable(stx_conv_float_oracle tests/float_oracle.cpp)
        target_include_directories(stx_conv_float_oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_link_libraries(stx_conv_float_oracle PRIVATE stx_conv::stx_conv)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            target_compile_options(stx_conv_float_oracle PRIVATE -ffp-contract=off)
        endif()
        add_test(NAME stx_conv_float_oracle COMMAND stx_conv_float_oracle)
        find_package(Threads REQUIRED)
        add_executable(stx_conv_comparison tests/comparison.cpp)
        target_include_directories(stx_conv_comparison PRIVATE "${STX_CONV_ORACLE}/src")
        target_link_libraries(stx_conv_comparison PRIVATE stx_conv::stx_conv Threads::Threads)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            target_compile_options(stx_conv_comparison PRIVATE -ffp-contract=off)
        endif()
        add_test(NAME stx_conv_oracle COMMAND stx_conv_comparison)
    endif()
endif()
install(TARGETS stx_conv EXPORT stx_convTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT stx_convTargets NAMESPACE stx_conv:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_conv)
configure_package_config_file(cmake/stx_convConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/stx_convConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_conv)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/stx_convConfigVersion.cmake
    VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/stx_convConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/stx_convConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_conv)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/stx_conv)
