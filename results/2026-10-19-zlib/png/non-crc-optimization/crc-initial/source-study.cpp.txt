crc_study::Update selected_crc = nullptr;
std::uint32_t upstream(std::span<const std::byte> bytes, std::uint32_t seed) {
    const Bytef* data = reinterpret_cast<const Bytef*>(bytes.data());
    const std::uint32_t result = static_cast<std::uint32_t>(crc32_z(seed, data, bytes.size()));
    return result;
}
std::uint32_t random_word(std::uint32_t& state) {
    state ^= state << 13;
    state ^= state >> 17;
    state ^= state << 5;
    return state;
}
std::vector<crc_study::Candidate> candidates() {
    std::vector<crc_study::Candidate> result{};
    constexpr std::array<unsigned, 6> indices{0,2,3,6,11,12};
    for (unsigned index : indices) result.push_back(crc_study::candidates[index]);
    for (const crc_study::Candidate& candidate : crc_suitability::circuits) result.push_back(candidate);
    result.push_back({"upstream-zlib", upstream, 0, 0, true});
    return result;
}
void verify() {
    std::uint32_t random = 0x20c0ffeeU;
    std::vector<std::byte> storage(1048576 + 32);
    for (std::byte& value : storage) value = static_cast<std::byte>(random_word(random) & 255U);
    std::vector<std::size_t> sizes{};
    for (std::size_t size = 0; size <= 321; ++size) sizes.push_back(size);
    for (std::size_t edge : {836U,1044U,1672U,4096U,65536U}) {
        for (std::size_t size = edge - 17; size <= edge + 17; ++size) sizes.push_back(size);
    }
    sizes.push_back(1048576);
    const std::vector<crc_study::Candidate> list = candidates();
    std::size_t checks = 0;
    for (std::size_t size : sizes) {
        for (unsigned alignment = 0; alignment < 16; ++alignment) {
            const std::span<const std::byte> bytes(storage.data() + alignment, size);
            const std::uint32_t seed = alignment % 2 == 0 ? 0 : random_word(random);
            const std::uint32_t expected = upstream(bytes, seed);
            if (size <= 321 && expected != crc_study::reference(bytes, seed)) throw std::runtime_error("independent oracle disagreement");
            for (const crc_study::Candidate& candidate : list) {
                if (!candidate.streaming && seed != 0) continue;
                const std::uint32_t actual = candidate.update(bytes, seed);
                if (actual != expected) {
                    std::cerr << candidate.name << " size=" << size << " alignment=" << alignment << '\n';
                    throw std::runtime_error("CRC mismatch");
                }
                ++checks;
            }
        }
    }
    // Exact allocations put ASan redzones immediately after each tested tail.
    for (std::size_t size : {208U,209U,835U,836U,837U,1043U,1044U,1045U,4097U}) {
        std::vector<std::byte> exact(size, std::byte{0xa5});
        for (const crc_study::Candidate& candidate : list) {
            const std::uint32_t expected = upstream(exact, 0);
            if (candidate.update(exact, 0) != expected) throw std::runtime_error("exact tail mismatch");
            if (!candidate.streaming) continue;
            for (std::size_t split = 0; split <= size; split += 17) {
                const std::span<const std::byte> bytes(exact);
                const std::uint32_t first = candidate.update(bytes.first(split), 0);
                const std::uint32_t actual = candidate.update(bytes.subspan(split), first);
                if (actual != expected) throw std::runtime_error("streaming mismatch");
                ++checks;
            }
        }
    }
    std::cout << "CRC checks: " << checks << "; independent bit oracle, pinned upstream, alignments, exact tails and split seeds passed\n";
}
struct Workload final {
    std::string name{};
    std::vector<std::byte> bytes{};
    std::vector<std::pair<std::size_t, std::size_t>> chunks{};
    std::size_t checked_bytes = 0;
};
std::uint32_t big32(const std::vector<std::byte>& bytes, std::size_t offset) {
    std::uint32_t value = 0;
    for (unsigned index = 0; index < 4; ++index) value = (value << 8) | std::to_integer<unsigned>(bytes[offset + index]);
    return value;
}
Workload read_png(const std::filesystem::path& path) {
    Workload work{};
    work.name = path.filename().string();
    const std::uintmax_t size = std::filesystem::file_size(path);
    if (size > 32ULL * 1024 * 1024) throw std::runtime_error("PNG too large");
    work.bytes.resize(static_cast<std::size_t>(size));
    std::ifstream file(path, std::ios::binary);
    file.read(reinterpret_cast<char*>(work.bytes.data()), static_cast<std::streamsize>(size));
    if (!file) throw std::runtime_error("PNG read failed");
    const gui_forms::image::PngDecodeResult inspected = gui_forms::image::inspect_png_samples(work.bytes, {});
    if (inspected.error != gui_forms::image::PngDecodeError::none) throw std::runtime_error("PNG validation failed");
    std::size_t offset = 8;
    while (offset < work.bytes.size()) {
        const std::size_t length = big32(work.bytes, offset);
        work.chunks.push_back({offset + 4, length + 4});
        work.checked_bytes += length + 4;
        offset += length + 12;
    }
    return work;
}
struct DecodeStorage final {
    std::unique_ptr<gui_forms::compression::InflateStream> workspace = std::make_unique<gui_forms::compression::InflateStream>();
    std::vector<std::byte> rows{};
    std::vector<std::uint16_t> rgba{};
    std::vector<std::byte> metadata{};
    std::vector<gui_forms::image::PngMetadataRecord> records{};
};
volatile std::uint32_t checksum_sink = 0;
double timed(const Workload& work, const crc_study::Candidate& candidate, unsigned iterations,
             bool decode, DecodeStorage& storage) {
    selected_crc = candidate.update;
    std::uint32_t sum = 0;
    const std::span<const std::byte> bytes(work.bytes);
    const std::chrono::steady_clock::time_point start = std::chrono::steady_clock::now();
    if (decode) {
        for (unsigned index = 0; index < iterations; ++index) {
            const gui_forms::image::PngDecodeResult result = gui_forms::image::crc_decode_png_samples(bytes, {},
                *storage.workspace, storage.rows, storage.rgba, storage.metadata, storage.records);
            if (result.error != gui_forms::image::PngDecodeError::none) throw std::runtime_error("candidate PNG decode failed");
            sum += storage.rgba[index % storage.rgba.size()];
        }
    } else {
        for (unsigned index = 0; index < iterations; ++index) {
            for (const std::pair<std::size_t, std::size_t>& chunk : work.chunks) {
                sum += candidate.update(bytes.subspan(chunk.first, chunk.second), 0);
            }
        }
    }
    const std::chrono::steady_clock::time_point end = std::chrono::steady_clock::now();
    checksum_sink = sum;
    const std::chrono::duration<double> duration = end - start;
    return duration.count();
}
void measure(const Workload& work, bool decode) {
    DecodeStorage storage{};
    if (decode) {
        const gui_forms::image::PngDecodeResult inspected = gui_forms::image::inspect_png_samples(work.bytes, {});
        storage.rows.resize(inspected.layout.row_storage_bytes);
        storage.rgba.resize(inspected.layout.rgba_samples);
        storage.metadata.resize(4 * 1024 * 1024);
        storage.records.resize(inspected.layout.metadata_records);
    }
    const std::vector<crc_study::Candidate> list = candidates();
    std::vector<unsigned> iterations(list.size(), 1);
    std::vector<std::array<double, 7>> samples(list.size());
    for (std::size_t mode = 0; mode < list.size(); ++mode) {
        while (timed(work, list[mode], iterations[mode], decode, storage) < 0.025) {
            if (iterations[mode] >= (1U << 24)) throw std::runtime_error("calibration cap");
            iterations[mode] *= 2;
        }
    }
    for (unsigned round = 0; round < 9; ++round) {
        for (std::size_t step = 0; step < list.size(); ++step) {
            const std::size_t position = round % 2 == 0 ? step : list.size() - 1 - step;
            const std::size_t mode = (position + round * 3) % list.size();
            const double duration = timed(work, list[mode], iterations[mode], decode, storage);
            if (round >= 2) samples[mode][round - 2] = duration * 1e9 / iterations[mode];
        }
    }
    for (std::size_t mode = 0; mode < list.size(); ++mode) {
        std::array<double, 7> ratios{};
        for (unsigned round = 0; round < 7; ++round) ratios[round] = samples[0][round] / samples[mode][round];
        std::array<double, 7> times = samples[mode];
        std::sort(times.begin(), times.end());
        std::sort(ratios.begin(), ratios.end());
        std::cout << (decode ? "png" : "crc") << ',' << work.name << ',' << list[mode].name << ','
            << work.checked_bytes << ',' << work.chunks.size() << ',' << iterations[mode] << ','
            << times[3] << ',' << times[0] << ',' << times[6] << ','
            << ratios[3] << ',' << ratios[0] << ',' << ratios[6] << '\n';
    }
    std::cout.flush();
}

std::uint32_t experimental_crc(std::span<const std::byte> bytes) noexcept {
    const std::uint32_t result = selected_crc(bytes, 0);
    return result;
}

int main(int argc, char** argv) {
    try {
        if (argc == 2 && std::string(argv[1]) == "verify") {
            verify();
            return 0;
        }
        if (argc != 3) return 2;
        const bool decode = std::string(argv[1]) == "png";
        std::cout << "mode,workload,candidate,crc_bytes,chunks,iterations,median_ns,min_ns,max_ns,paired_speedup,paired_min_speedup,paired_max_speedup\n";
        if (!decode) {
            std::uint32_t random = 42;
            for (std::size_t size : {4U,17U,64U,208U,512U,1044U,4096U,16384U,65539U,262144U,1048576U}) {
                Workload work{};
                work.name = "random-" + std::to_string(size);
                work.bytes.resize(size);
                for (std::byte& value : work.bytes) value = static_cast<std::byte>(random_word(random) & 255U);
                work.chunks.push_back({0, size});
                work.checked_bytes = size;
                measure(work, false);
            }
        }
        std::vector<std::filesystem::path> paths{};
        for (const std::filesystem::directory_entry& entry : std::filesystem::directory_iterator(argv[2])) {
            if (entry.path().extension() == ".png") paths.push_back(entry.path());
        }
        std::sort(paths.begin(), paths.end());
        for (const std::filesystem::path& path : paths) {
            const Workload work = read_png(path);
            measure(work, decode);
        }
        return 0;
    } catch (const std::exception& error) {
        std::cerr << error.what() << '\n';
        return 1;
    }
}
