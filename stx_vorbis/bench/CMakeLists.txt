# Opt-in measurement executables; external reference sources remain outside Git.
add_executable(stx_bench decode.cpp)
target_link_libraries(stx_bench PRIVATE stx_vorbis)
stx_warnings(stx_bench)
if(STX_VORBIS_REFERENCE)
  add_executable(stx_bench_libvorbis decode.cpp)
  target_compile_features(stx_bench_libvorbis PRIVATE cxx_std_20)
  target_compile_definitions(stx_bench_libvorbis PRIVATE STX_BENCH_LIBVORBIS)
  target_link_libraries(stx_bench_libvorbis PRIVATE PkgConfig::VORBIS)
  stx_warnings(stx_bench_libvorbis)
endif()
set(STX_BENCH_STB_SOURCE "" CACHE FILEPATH "Pinned external stb_vorbis.c, benchmark only")
if(STX_BENCH_STB_SOURCE)
  add_executable(stx_bench_stb decode.cpp ${STX_BENCH_STB_SOURCE})
  target_compile_features(stx_bench_stb PRIVATE cxx_std_20)
  target_compile_definitions(stx_bench_stb PRIVATE STX_BENCH_STB)
  target_include_directories(stx_bench_stb PRIVATE ../include)
endif()
set(STX_BENCH_TREMOR_ROOT "" CACHE PATH "Pinned external Tremor source directory, benchmark only")
if(STX_BENCH_TREMOR_ROOT)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(OGG REQUIRED IMPORTED_TARGET ogg)
  set(tremor_sources)
  foreach(name mdct block window synthesis info floor1 floor0 vorbisfile res012 mapping0 registry codebook sharedbook)
    list(APPEND tremor_sources ${STX_BENCH_TREMOR_ROOT}/${name}.c)
  endforeach()
  add_executable(stx_bench_tremor decode.cpp ${tremor_sources})
  target_compile_features(stx_bench_tremor PRIVATE cxx_std_20)
  target_compile_definitions(stx_bench_tremor PRIVATE STX_BENCH_TREMOR)
  target_include_directories(stx_bench_tremor PRIVATE ${STX_BENCH_TREMOR_ROOT})
  target_link_libraries(stx_bench_tremor PRIVATE PkgConfig::OGG)
endif()

if(STX_VORBIS_TESTS)
  add_test(NAME stx_vorbis_benchmark_smoke COMMAND stx_bench
    ${CMAKE_CURRENT_SOURCE_DIR}/../tests/corpus/floor1-residue2.ogg 2 1 scalar)
  if(STX_VORBIS_REFERENCE)
    add_test(NAME stx_vorbis_reference_benchmark_smoke COMMAND stx_bench_libvorbis
      ${CMAKE_CURRENT_SOURCE_DIR}/../tests/corpus/floor1-residue2.ogg 2 1)
  endif()
endif()
